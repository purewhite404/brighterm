const gallery = document.getElementById('gallery')
const status = document.getElementById('status')
const viewer = document.getElementById('viewer')
const largeImage = document.getElementById('largeImage')

async function show(folder) {
  gallery.innerHTML = ''
  const entries = await window.brighterm.fs.listFiles(folder)
  const photos = entries.filter((e) => !e.isDirectory && /\.(jpe?g|png|gif|webp|bmp|avif)$/i.test(e.name))
  status.textContent = photos.length ? `${folder.label}: ${photos.length} 枚` : '画像が見つかりませんでした。'
  for (const photo of photos) {
    const img = document.createElement('img')
    img.loading = 'lazy'
    img.alt = photo.name
    img.src = await window.brighterm.fs.fileUrl(folder, photo.name)
    img.addEventListener('click', () => {
      largeImage.src = img.src
      viewer.showModal()
    })
    gallery.append(img)
  }
}

document.getElementById('pickButton').addEventListener('click', async () => {
  const folder = await window.brighterm.fs.pickFolder()
  if (!folder) return
  await window.brighterm.storage.set('folder', folder)
  await show(folder)
})

viewer.addEventListener('click', () => viewer.close())

window.brighterm.storage.get('folder').then((folder) => folder && show(folder))
