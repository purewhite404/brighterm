=== file: main.js ===
const pickButton = document.getElementById("pickButton");
const gallery = document.getElementById("gallery");
const status = document.getElementById("status");
const viewer = document.getElementById("viewer");
const largeImage = document.getElementById("largeImage");
const closeButton = document.getElementById("closeButton");

const imageExtensions = [
  "jpg",
  "jpeg",
  "png",
  "gif",
  "webp",
  "bmp"
];

let currentFolder = null;

pickButton.addEventListener("click", async () => {
  try {
    currentFolder = await window.brighterm.fs.pickFolder();

    if (!currentFolder) {
      return;
    }

    await loadPhotos();
  } catch (error) {
    status.textContent = "フォルダを開けませんでした。";
  }
});

async function loadPhotos() {
  gallery.innerHTML = "";
  status.textContent = "読み込み中...";

  try {
    const files = await window.brighterm.fs.listFiles(currentFolder);
    const photos = files.filter(file => {
      const parts = file.name.toLowerCase().split(".");
      return parts.length > 1 &&
        imageExtensions.includes(parts.pop());
    });

    if (photos.length === 0) {
      status.textContent = "画像が見つかりませんでした。";
      return;
    }

    status.textContent = `${photos.length} 枚の写真`;

    for (const photo of photos) {
      const data = await window.brighterm.fs.readFile(photo.path);
      const url = createImageUrl(data);

      const item = document.createElement("button");
      item.className = "photo-card";

      const image = document.createElement("img");
      image.src = url;
      image.alt = photo.name;

      item.appendChild(image);
      item.addEventListener("click", () => {
        largeImage.src = url;
        viewer.showModal();
      });

      gallery.appendChild(item);
    }
  } catch (error) {
    status.textContent = "写真の読み込みに失敗しました。";
  }
}

function createImageUrl(data) {
  if (typeof data === "string") {
    return data;
  }

  const blob = new Blob([data]);
  return URL.createObjectURL(blob);
}

closeButton.addEventListener("click", () => {
  viewer.close();
});
